R1 

configure terminal

hostname R1
no ip domain lookup
ip routing

enable secret cisco

interface Serial0/1/0
 ip address 10.0.0.1 255.255.255.252
 clock rate 64000
 no shutdown

interface GigabitEthernet0/0/0
 ip address 192.168.1.1 255.255.255.0
 no shutdown

ip route 192.168.2.0 255.255.255.0 10.0.0.2

ip domain name lab.local
username admin privilege 15 secret cisco
crypto key generate rsa modulus 2048
ip ssh version 2

line console 0
 password cisco
 login
 exec-timeout 10 0
 exit

line vty 0 15
 login local
 transport input ssh
 exec-timeout 10 0
 exit

end
write memory
